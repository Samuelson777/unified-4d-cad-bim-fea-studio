// ============================================================================
// VERTEX SHADER (OpenGL 3.3 Core / WebGL 2.0 ES 300)
// ============================================================================
#version 300 es
precision highp float;

layout(location = 0) in vec4 a_position4D;     // (x, y, z, w) in R^4
layout(location = 1) in vec3 a_barycentric;    // (1,0,0), (0,1,0), or (0,0,1)
layout(location = 2) in float a_vonMisesStress;// Normalized nodal stress [0.0, 1.0]

uniform mat4 u_rotationSO4;                    // 4x4 SO(4) orthogonal rotation matrix
uniform float u_eyeW;                          // 4D camera w-distance (e.g., 3.0)
uniform int u_projectionMode;                  // 0 = Perspective, 1 = Orthographic
uniform mat4 u_modelViewProjection3D;          // Standard 3D camera MVP matrix

out vec3 v_barycentric;
out float v_stress;
out float v_wCoord;

void main() {
    // 1. Rotate vertex in 4D Euclidean space via SO(4) matrix
    vec4 rotated4D = u_rotationSO4 * a_position4D;
    v_wCoord = rotated4D.w;

    // 2. Project from 4D (x, y, z, w) to 3D shadow (x', y', z')
    float scaleFactor = (u_projectionMode == 0) ? (u_eyeW / max(0.2, u_eyeW - rotated4D.w)) : 1.0;
    vec3 shadow3D = rotated4D.xyz * scaleFactor;

    // 3. Project 3D shadow to 2D clip space
    gl_Position = u_modelViewProjection3D * vec4(shadow3D, 1.0);
    v_barycentric = a_barycentric;
    v_stress = a_vonMisesStress;
}

// ============================================================================
// FRAGMENT SHADER (Single-Pass Shortest-Edge Distance Wireframe + FEA Jet Map)
// ============================================================================
#version 300 es
precision highp float;

in vec3 v_barycentric;
in float v_stress;
in float v_wCoord;

uniform float u_wireWidthPx;                   // Target line thickness in screen pixels
uniform int u_renderMode;                      // 0 = Solid, 1 = Wireframe, 2 = Solid+Wire, 3 = VonMises
uniform vec4 u_wireframeColor;
uniform vec4 u_solidFillColor;

out vec4 fragColor;

vec3 evaluateJetColormap(float t) {
    float v = clamp(t, 0.0, 1.0);
    return clamp(vec3(1.5 - abs(4.0 * v - 3.0),
                      1.5 - abs(4.0 * v - 2.0),
                      1.5 - abs(4.0 * v - 1.0)), 0.0, 1.0);
}

void main() {
    // Screen-space rate of change of barycentric coordinates via hardware derivatives
    vec3 dBary = fwidth(v_barycentric);
    vec3 edgeDistancesPx = v_barycentric / max(dBary, vec3(1e-6));
    float minDistanceToEdgePx = min(edgeDistancesPx.x, min(edgeDistancesPx.y, edgeDistancesPx.z));

    // Antialiased smoothstep edge factor (1.0 on edge, 0.0 in triangle interior)
    float edgeIntensity = 1.0 - smoothstep(u_wireWidthPx - 1.0, u_wireWidthPx + 1.0, minDistanceToEdgePx);

    vec4 baseSurfaceColor = (u_renderMode == 3)
        ? vec4(evaluateJetColormap(v_stress), 0.90)
        : u_solidFillColor;

    if (u_renderMode == 1) {
        if (edgeIntensity < 0.05) discard;
        fragColor = vec4(u_wireframeColor.rgb, edgeIntensity);
    } else if (u_renderMode == 2 || u_renderMode == 3) {
        fragColor = mix(baseSurfaceColor, u_wireframeColor, edgeIntensity);
    } else {
        fragColor = baseSurfaceColor;
    }
}
